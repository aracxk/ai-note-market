+# ADR 0008: 読み取り専用モデル（QueryService）としての CQRS パターンの採用
+
+## 日付
+2026-09-11
+
+## ステータス
+Accepted
+
+## 背景と課題
+記事の一覧（トップページ）やマイページの購入履歴一覧を表示する際、これまでの `Repository` を使って `Note` や `Purchase` などの集約（エンティティ）を再構築（復元）してから画面に渡すと、以下の問題が発生します。
+
+1. **パフォーマンスとメモリの浪費**: 画面には「タイトル」と「価格」しか必要ないのに、本文（Content）などの重いデータをすべてメモリにロードしてエンティティを作ってしまう。
+2. **N+1問題の発生**: 購入履歴から「購入した記事のタイトル」を表示する際、Purchase リポジトリで取得した後に Note リポジトリをループで叩くことになり、N+1クエリが発生する。
+3. **副作用の危険性**: 画面表示のためだけにエンティティをロードすると、誤ってエンティティのメソッド（書き込み処理）を呼び出して状態を変更してしまうリスクがある。
+
+## 決定事項
+「書き込み用のドメインモデル（Repository）」と「読み取り用のクエリモデル（QueryService）」を明確に分離する **CQRS (Command Query Responsibility Segregation)** パターンを採用します。
+
+1. **QueryServiceの導入**:
+   - 画面用のデータ取得専用のインターフェースとして `QueryService` を定義する。
+   - インターフェースの定義は `src/features/<feature>/queries/` に配置する。
+2. **DTO (Data Transfer Object) の利用**:
+   - `QueryService` の戻り値は、ドメインエンティティではなく、プリミティブな型のみを持つ `readonly` な DTO とする。
+3. **インフラ層での直接JOIN**:
+   - 実装（`infrastructure`）側では、ORマッパーなどを駆使してデータベース側で JOIN などを効率的に行い、直接 DTO を生成して返す（現在はインメモリで疑似的にJOINしている）。
+
+## 影響
+- **ポジティブ**:
+  - 画面表示のためのクエリが劇的に高速化し、メモリ使用量も減る。
+  - `Repository` は「ドメインの書き込み・整合性維持」のみに集中でき、インターフェースがシンプルになる。
+- **ネガティブ**:
+  - `Repository` と `QueryService` の2種類のインターフェースと実装が存在するため、ファイル数が増える。
+
+## 備考
+サブエージェント（レビューAI）の指摘（E-3）に基づき起票しました。

+import { NOTE_STATUS } from "../../note/domain/NoteStatus";
+import type {
+	INoteSummaryQueryService,
+	NoteSummaryDto,
+} from "../queries/INoteSummaryQueryService";
+import type { InMemoryNoteRepository } from "./InMemoryNoteRepository";
+
+export class InMemoryNoteSummaryQueryService
+	implements INoteSummaryQueryService
+{
+	constructor(private readonly noteRepository: InMemoryNoteRepository) {}
+
+	public async findPublishedNotes(
+		limit?: number,
+		offset?: number,
+	): Promise<NoteSummaryDto[]> {
+		// 1. メモリ上の全記事を取得
+		const allNotes = this.noteRepository.getAll();
+
+		// 2. 公開中の記事だけをフィルタリングし、DTOに変換
+		const publishedNotes = allNotes
+			.filter((note) => note.status === NOTE_STATUS.PUBLISHED)
+			.map((note) => ({
+				noteId: note.id.value,
+				authorId: note.authorId.value,
+				title: note.title.value,
+				price: note.price.amount,
+				category: `${note.category.major}/${note.category.minor}`,
+				publishedAt: note.updatedAt, // 公開日時は最終更新日時を代用
+			}));
+
+		// 3. 公開日時の降順（新しい順）でソート
+		publishedNotes.sort(
+			(a, b) => b.publishedAt.getTime() - a.publishedAt.getTime(),
+		);
+
+		// 4. ページネーション（limit / offset）を適用
+		const start = offset ?? 0;
+		const end = limit !== undefined ? start + limit : undefined;
+
+		return publishedNotes.slice(start, end);
+	}
+}

+import type { InMemoryNoteRepository } from "../../note/infrastructure/InMemoryNoteRepository";
+import type {
+	IPurchaseHistoryQueryService,
+	PurchaseHistoryDto,
+} from "../queries/IPurchaseHistoryQueryService";
+import type { InMemoryPurchaseRepository } from "./InMemoryPurchaseRepository";
+
+export class InMemoryPurchaseHistoryQueryService
+	implements IPurchaseHistoryQueryService
+{
+	constructor(
+		private readonly purchaseRepository: InMemoryPurchaseRepository,
+		private readonly noteRepository: InMemoryNoteRepository,
+	) {}
+
+	public async findByBuyerId(buyerId: string): Promise<PurchaseHistoryDto[]> {
+		// 1. 購入者の購入履歴を全件取得
+		const purchases = this.purchaseRepository
+			.getAll()
+			.filter((p) => p.buyerId.value === buyerId);
+
+		const dtos: PurchaseHistoryDto[] = [];
+		const allNotes = this.noteRepository.getAll();
+
+		// 2. 購入履歴と記事情報をインメモリで手動 JOIN する
+		for (const purchase of purchases) {
+			const note = allNotes.find((n) => n.id.equals(purchase.noteId));
+			if (!note) {
+				continue; // 記事が見つからない場合はスキップ（実際のDBでは外部キー制約で防ぐ）
+			}
+
+			dtos.push({
+				purchaseId: purchase.id.value,
+				noteId: note.id.value,
+				noteTitle: note.title.value,
+				authorId: note.authorId.value,
+				price: purchase.purchasedPrice.amount, // 購入時の価格スナップショットを使う
+				purchasedAt: purchase.purchasedAt,
+			});
+		}
+
+		// 3. 購入日時の降順（新しい順）でソート
+		dtos.sort((a, b) => b.purchasedAt.getTime() - a.purchasedAt.getTime());
+
+		return dtos;
+	}
+}
